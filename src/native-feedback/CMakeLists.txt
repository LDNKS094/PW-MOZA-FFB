cmake_minimum_required(VERSION 3.24)
project(PWNativeFeedback LANGUAGES C CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 is required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
FetchContent_Declare(minhook
  GIT_REPOSITORY https://github.com/TsudaKageyu/minhook.git
  GIT_TAG c3fcafdc10146beb5919319d0683e44e3c30d537)
FetchContent_MakeAvailable(minhook)
add_library(PWNativeFeedback SHARED adapter.cpp)
target_link_libraries(PWNativeFeedback PRIVATE minhook bcrypt dinput8 dxguid)
foreach(target PWNativeFeedback)
  target_compile_definitions(${target} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_compile_options(${target} PRIVATE /W4 /permissive- /EHsc)
endforeach()
install(TARGETS PWNativeFeedback RUNTIME DESTINATION . COMPONENT Runtime)
install(FILES README.md THIRD-PARTY.md DESTINATION . COMPONENT Runtime)
install(FILES "${minhook_SOURCE_DIR}/LICENSE.txt" DESTINATION . RENAME LICENSE-Minhook.txt COMPONENT Runtime)
install(FILES "${CMAKE_CURRENT_SOURCE_DIR}/../../LICENSE" DESTINATION . RENAME LICENSE.txt COMPONENT Runtime)
